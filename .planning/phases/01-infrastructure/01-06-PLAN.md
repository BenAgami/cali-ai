```typescript
export class TimeoutError extends Error {
  constructor(readonly code: string, readonly timeoutMs: number) {
    super(`${code}: timed out after ${timeoutMs} ms`);
    this.name = "TimeoutError";
  }
}

export const withTimeout = <T>(promise: Promise<T>, timeoutMs: number, code: string): Promise<T> =>
  new Promise<T>((resolve, reject) => {
    const timer = setTimeout(() => reject(new TimeoutError(code, timeoutMs)), timeoutMs);
    promise.then(
      (value) => { clearTimeout(timer); resolve(value); },
      (error: unknown) => { clearTimeout(timer); reject(error instanceof Error ? error : toError(error)); },
    );
  });
```
(Import `toError` from `./toError`.) Note in a doc comment that the inner work is not cancelled — callers must make late results harmless (the worker's guarded writes do).
