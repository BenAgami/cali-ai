```ts
// apps/native/src/pose/types.ts
export type PoseKeypoint = [x: number, y: number, visibility: number];
/** 33 BlazePose landmarks, x/y normalised 0..1 to the upright (oriented) image, y down, NOT mirrored. */
export type PoseFrame = { timestampSec: number; keypoints: PoseKeypoint[] };
export type PoseSourceInfo = {
  poseModelName: "blazepose_full";
  poseModelVersion: string;   // e.g. "pose_landmarker_full.task@<plugin version>"
  aspectRatio: number;        // width / height of the oriented image the landmarks are normalised to
};
export type PoseSource = {
  isReady: boolean;
  info: PoseSourceInfo | null;
  frameProcessor: ReadonlyFrameProcessor | undefined; // import type { ReadonlyFrameProcessor } from "react-native-vision-camera"
};
export type UsePoseSourceOptions = {
  isActive: boolean;
  cameraPosition: "front" | "back";
  targetFps: number;          // 15 for recording
  onFrame: (frame: PoseFrame) => void; // invoked on the JS thread via createRunOnJS
};
```
