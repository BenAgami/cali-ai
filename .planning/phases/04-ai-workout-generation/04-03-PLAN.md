`bodyMetricService.ts`:
```ts
const roundOneDecimal = (value: number): number => Math.round(value * 10) / 10;
export type LatestBodyMetric = { heightCm: number; weightKg: number; createdAt: Date };
export class BodyMetricService {
  private get prisma() { return getPrismaClient(); }
  async getLatest(userId: number): Promise<LatestBodyMetric | null> { /* findFirst as in behavior */ }
  async saveIfChanged(userUuid: string, input: BodyMetricInput): Promise<{ bodyMetric: LatestBodyMetric; created: boolean }> {
    const userId = await getUserIdByUuid(this.prisma, userUuid);
    const heightCm = roundOneDecimal(input.heightCm);
    const weightKg = roundOneDecimal(input.weightKg);
    const latest = await this.getLatest(userId);
    if (latest && latest.heightCm === heightCm && latest.weightKg === weightKg) return { bodyMetric: latest, created: false };
    const bodyMetric = await this.prisma.bodyMetric.create({ data: { userId, heightCm, weightKg }, select: { heightCm: true, weightKg: true, createdAt: true } });
    return { bodyMetric, created: true };
  }
}
export default new BodyMetricService();
```
`aiPrerequisitesService.ts` (the single shared gate, D-05; Phase 5 will call the same method):
```ts
export type AiUserContext = { userId: number; experienceLevel: ExperienceLevel; goal: { goalType: GoalType; title: string }; heightCm: number; weightKg: number };
const MESSAGES: Record<AiPrerequisiteCode, string> = {
  PROFILE_INCOMPLETE: "Set your fitness level and primary goal before using AI features",
  BODY_METRICS_REQUIRED: "Add your height and weight before using AI features",
  AI_CONSENT_REQUIRED: "Accept the AI privacy notice before using AI features",
};
export class AiPrerequisitesService {
  private get prisma() { return getPrismaClient(); }
  async assertAiPrerequisites(userId: number): Promise<AiUserContext> {
    const user = await this.prisma.user.findUnique({ where: { id: userId }, select: { experienceLevel: true, aiConsentVersion: true, goals: { where: { status: "ACTIVE" }, orderBy: { createdAt: "desc" }, take: 1, select: { goalType: true, title: true } } } });
    if (!user) throw new NotFoundError("User not found");
    const bodyMetric = await bodyMetricService.getLatest(userId);
    const goal = user.goals[0];
    const missing = getMissingAiPrerequisites({ experienceLevel: user.experienceLevel, hasActiveGoal: goal !== undefined, hasBodyMetric: bodyMetric !== null, aiConsentVersion: user.aiConsentVersion });
    if (missing.length > 0) { const code = missing[0]; throw new PreconditionRequiredError(MESSAGES[code], code); }
    // narrowings are guaranteed by the check above
    return { userId, experienceLevel: user.experienceLevel!, goal: { goalType: goal!.goalType, title: goal!.title }, heightCm: bodyMetric!.heightCm, weightKg: bodyMetric!.weightKg };
  }
}
export default new AiPrerequisitesService();
```
