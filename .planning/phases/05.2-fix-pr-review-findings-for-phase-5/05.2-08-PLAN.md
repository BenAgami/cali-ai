async generateFormFeedback(input: { metrics: ScoredFormMetrics; context: FormFeedbackContext; signal: AbortSignal; timeoutMs: number; log: Logger }): Promise<FormFeedbackOutcome> {
  const deadline = Date.now() + input.timeoutMs;
  for (let attempt = 1; attempt <= MAX_FORM_ATTEMPTS; attempt++) {
    const remaining = deadline - Date.now();
    if (remaining < MIN_ATTEMPT_MS) throw new LlmError("TIMEOUT", "Form feedback deadline reached");
    ...getLlmProvider().generateFormFeedback({ model, effort, maxTokens, timeoutMs: remaining, signal, system, userPrompt, allowedTags })
    catch (error) { if (error instanceof LlmError && (error.code === "REFUSAL" || error.code === "INVALID_OUTPUT")) { previousIssues = [GENERIC_FORM_ISSUE]; continue; } throw error; }
    ...validation: on failure previousIssues = validation.issues; log.warn({ attempt, issues }, "Form feedback failed validation");
  }
  throw new LlmError("INVALID_OUTPUT", "Form feedback failed validation after repair");
}
