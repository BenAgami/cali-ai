```typescript
const hadMotion = (frames: readonly SetupFrame[]): boolean => {
  const first = frames[0];
  const last = frames[frames.length - 1];
  if (!first || !last) return false;
  let lo = 0;
  let hi = 0;
  for (let start = first.timestampSec; start + ACTIVE_WINDOW_SECS <= last.timestampSec; start += 1) {
    while (lo < frames.length && frames[lo].timestampSec < start) lo++;
    if (hi < lo) hi = lo;
    while (hi < frames.length && frames[hi].timestampSec < start + ACTIVE_WINDOW_SECS) hi++;
    if (motionRange(frames.slice(lo, hi)) >= ACTIVE_RANGE) return true;
  }
  return false;
};
```
