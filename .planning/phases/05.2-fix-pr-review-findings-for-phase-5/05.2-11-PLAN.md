  const stored = await this.prisma.poseCapture.findFirst({ where: { sessionId, keypoints: { not: null } }, select: { id: true } });

  await this.publishSnapshot(sessionId, logger);
  return { captureId: stored.id };
}

```typescript
async restartRun(sessionId: number, jobId: string, userId: number): Promise<{ captureId: number }> {
  const stored = await this.prisma.poseCapture.findFirst({ where: { sessionId, keypoints: { not: null } }, select: { id: true } });
  if (!stored) throw new ConflictError("No stored capture to retry for this session");

  await this.prisma.$transaction(async (tx) => {
    await this.lockUserQuota(tx, userId);
    const acceptedRetryCount = await tx.analysisRetry.count({
      where: { session: { userId }, createdAt: { gte: dayAgo() } },
    });
    if (acceptedRetryCount >= env.formAnalysis.dailyRetryCap) {
      throw new TooManyRequestsError(ANALYSIS_LIMIT_MESSAGES.dailyRetry, API_ERROR_CODES.DAILY_RETRY_LIMIT_REACHED);
    }
    const { count } = await tx.workoutSession.updateMany({
      where: { id: sessionId, processingStatus: SessionStatus.FAILED },
      data: this.runResetData(jobId),
    });
    if (count === 0) throw new ConflictError("Only failed analyses can be retried");
    await tx.analysisRetry.create({ data: { sessionId } });
  });

  await this.publishSnapshot(sessionId, logger);
  return { captureId: stored.id };
}
```
